#!/usr/bin/env python3
"""One isolated, sequential TASK617 experiment; not an acceptance gate."""
import argparse
import hashlib
import datetime
import json
import os
from pathlib import Path
import platform
import re
import signal
import subprocess
import time

PIN = '3fa577340ededf26fcb2275733be8b625e63b876'
GO = Path('/Users/archmagece/.local/share/mise/installs/go/1.27.3')
DVA = Path.home() / '.local/share/ce/bin/dva'


def query(argv, cwd, env=None):
    return subprocess.check_output(argv, cwd=cwd, env=env, text=True, timeout=20).strip()


def status(root, env):
    try:
        value = json.loads(query([str(DVA), 'ci', 'status'], root, env))
        return value if isinstance(value, list) else []
    except (ValueError, subprocess.SubprocessError, OSError):
        return []


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--root', required=True, type=Path)
    parser.add_argument('--case', required=True)
    parser.add_argument('--output', required=True, type=Path)
    parser.add_argument('--repeat-default', action='store_true')
    args = parser.parse_args()
    root = args.root.resolve(strict=True)
    if not re.fullmatch(r'[A-Za-z0-9_-]+', args.case):
        raise ValueError('case must be a simple identifier')
    if query(['git', 'rev-parse', 'HEAD'], root) != PIN:
        raise ValueError('unexpected HEAD')
    if not (GO / 'bin/go').is_file() or (GO / 'VERSION').read_text().splitlines()[0] != 'go1.27.3':
        raise ValueError('pinned Go binary or VERSION missing')
    version = query([str(DVA), 'version'], root)
    if 'commit: ' + PIN not in version.splitlines():
        raise ValueError('installed DVA commit mismatch')
    cache = root / 'tmp' / 'task617-cache'
    if args.repeat_default:
        assert cache.is_dir(), 'existing measured cache required'
    else:
        cache.mkdir(parents=False, exist_ok=False)
    output = args.output.resolve()
    if args.repeat_default:
        assert output.is_dir() and not (output / 'warm-default-repeat-aligned.stdout').exists()
    else:
        output.mkdir(parents=True, exist_ok=False)
    env = os.environ.copy()
    env.update(GOROOT=str(GO), GOCACHE=str(cache), GOPROXY='off', GOSUMDB='off', GOTOOLCHAIN='local', PATH=str(GO / 'bin') + os.pathsep + env.get('PATH', ''))
    staged = query(['git', 'diff', '--cached', '--binary', '--no-ext-diff'], root)
    (output / 'staged.diff').write_text(staged + ('\n' if staged else ''))
    metadata = {'case': args.case, 'root': str(root), 'head': PIN, 'dva_version': version,
                'go_version_file': (GO / 'VERSION').read_text(), 'cpu': platform.processor(),
                'cpu_count': os.cpu_count(), 'memory_bytes': int(query(['/usr/sbin/sysctl', '-n', 'hw.memsize'], root)),
                'owner': 'codex/mbp', 'task': 'TASK-617', 'purpose': 'diagnostic, not acceptance',
                'created_at': datetime.datetime.now(datetime.timezone.utc).isoformat(),
                'cleanup': 'record evidence, then CE checkpoint/discard owned unpublished experiment',
                'cache_initial': 'new empty directory', 'cache_path': str(cache),
                'environment': {k: env.get(k) for k in ('GOROOT', 'GOCACHE', 'GOPROXY', 'GOSUMDB', 'GOTOOLCHAIN', 'GOFLAGS', 'GOMAXPROCS')},
                'path_policy': 'same declared Go bin first; remaining inherited PATH unchanged',
                'path_sha256': hashlib.sha256(env['PATH'].encode()).hexdigest(),
                'module_cache': 'existing, unchanged by harness', 'staged_diff': str(output / 'staged.diff'), 'runs': []}
    if args.repeat_default:
        metadata = json.loads((output / 'results.json').read_text())
        metadata['aligned_environment'] = {'GOROOT': str(GO), 'Go pin': '1.27.3', 'scope': 'per-run fixture only; no workstation changes'}
        metadata['repeat_reason'] = 'Observe existing Go test-result cache after the first non-fresh run; fresh runs do not seed result-cache evidence.'
    conditions = [('warm-default-repeat-aligned', False)] if args.repeat_default else [('cold-fresh', True), ('warm-fresh', True), ('warm-default', False)]
    for label, fresh in conditions:
        argv = [str(DVA), 'ci', 'commit'] + (['--fresh'] if fresh else []) + ['--json']
        before = {r.get('id') for r in status(root, env)}
        started = datetime.datetime.now(datetime.timezone.utc)
        start = time.monotonic()
        stdout = output / (label + '.stdout')
        stderr = output / (label + '.stderr')
        timed_out = False
        with stdout.open('wb') as out, stderr.open('wb') as err:
            proc = subprocess.Popen(argv, cwd=root, env=env, stdout=out, stderr=err, start_new_session=True)
            try:
                rc = proc.wait(timeout=170)
            except subprocess.TimeoutExpired:
                timed_out = True
                os.killpg(proc.pid, signal.SIGTERM)
                try:
                    rc = proc.wait(timeout=5)
                except subprocess.TimeoutExpired:
                    os.killpg(proc.pid, signal.SIGKILL)
                    rc = proc.wait()
        run = {'cache_before': ('empty' if not metadata['runs'] else ('previous execution succeeded' if metadata['runs'][-1]['returncode'] == 0 else 'partially populated after a failed execution')), 'condition': label, 'argv': argv, 'wall_seconds': time.monotonic() - start,
               'returncode': rc, 'outer_timeout': timed_out, 'stdout': str(stdout), 'stderr': str(stderr),
               'cache_may_be_partial': rc != 0 or timed_out}
        try:
            report = json.loads(stdout.read_text())
            if not isinstance(report, dict) or report.get('root') != str(root) or report.get('profile') != 'commit':
                report = None
        except (ValueError, OSError):
            report = None
        if report is None:
            candidates = []
            for r in status(root, env):
                try:
                    when = datetime.datetime.fromisoformat(r['started_at'].replace('Z', '+00:00'))
                    if r.get('id') not in before and r.get('root') == str(root) and r.get('profile') == 'commit' and when >= started:
                        candidates.append(r)
                except (KeyError, ValueError, TypeError):
                    continue
            if len(candidates) == 1:
                report = candidates[0]
                run['report_source'] = 'unique new status receipt'
            else:
                run['report_unavailable'] = 'no unique new matching receipt'
        else:
            run['report_source'] = 'stdout JSON'
        if report:
            run['report'] = report
            log = report.get('log_path')
            if log:
                try:
                    lines = Path(log).read_text(errors='replace').splitlines()
                    run['package_lines'] = [line for line in lines if re.search(r'(^|\s)(ok|\?|FAIL)\s+github\.com/', line)]
                except OSError as exc:
                    run['log_error'] = str(exc)
        metadata['runs'].append(run)
        (output / 'results.json').write_text(json.dumps(metadata, indent=2) + '\n')


if __name__ == '__main__':
    main()
