"""TASK-619 bounded diagnostic, never a replacement for final CI."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import time

root = Path.cwd()
dest = root / "tmp/task619/overhead"
dest.mkdir(mode=0o700)
state = dest / "metrics"
state.mkdir(mode=0o700)
env = dict(os.environ, GOCACHE=os.environ.get("GOCACHE", str(Path.home() / "Library/Caches/go-build")),
           GOPROXY="off", GOSUMDB="off", GOTOOLCHAIN="local", GOMAXPROCS="1", GOFLAGS="-p=1",
           DVA_CI_JOBS="4", DVA_CI_FRESH="1", DVA_PROJECT_CI_METRICS_DIR=str(state))
base = ["go", "test", "-count=1", "-timeout", "75s", "./internal/agentdeny"]
observed = ["go", "run", "./tools/cimetrics", "run", "--check", "commit-test", "--",
            "go", "test", "-json", *base[2:]]
def output(cmd):
    return subprocess.check_output(cmd, cwd=root, text=True).strip()
paths = set(output(["git", "diff", "--name-only"]).splitlines())
paths.update(output(["git", "ls-files", "--others", "--exclude-standard"]).splitlines())
code = {p: hashlib.sha256((root / p).read_bytes()).hexdigest() for p in sorted(paths)
        if p.endswith(".go") or p in {"Makefile", "dva.yml", ".github/workflows/ci.yml", "scripts/ci-lint.sh"}}
result = {"owner": "codex/mbp", "task": "TASK-619", "purpose": "diagnostic, not final acceptance",
          "created_at": datetime.datetime.now(datetime.timezone.utc).isoformat(),
          "baseline": output(["git", "rev-parse", "HEAD"]), "working_code_sha256": code,
          "conditions": "prepared module cache; existing shared Go build cache; recorded warmups; test result cache disabled by -count=1; other host load uncontrolled",
          "environment": {k: env[k] for k in ["GOPROXY", "GOSUMDB", "GOTOOLCHAIN", "GOMAXPROCS", "GOFLAGS", "DVA_CI_JOBS", "DVA_CI_FRESH"]},
          "declared_toolchain": "go1.27.3", "selected_go_version": "see wrapper runtime/build observations; a shim is not assumed to prove the selected Go version",
          "limits": "90s per command; own process group TERM then KILL; only sanitized metrics copied to evidence",
          "cleanup": "CE delivery worktree reclamation after preserving diagnostic summary", "runs": []}
for index, mode in enumerate(["warmup-direct", "warmup-observed", "direct", "observed", "observed", "direct", "direct", "observed"]):
    cmd = observed if mode.endswith("observed") else base
    old = set(state.glob("*/*.json"))
    outpath, errpath = dest / f"{index}.stdout", dest / f"{index}.stderr"
    start = time.perf_counter()
    timed_out = False
    with os.fdopen(os.open(outpath, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "wb") as out, \
         os.fdopen(os.open(errpath, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "wb") as err:
        proc = subprocess.Popen(cmd, cwd=root, env=env, stdout=out, stderr=err, start_new_session=True)
        try:
            code_rc = proc.wait(timeout=90)
        except subprocess.TimeoutExpired:
            timed_out = True
            os.killpg(proc.pid, signal.SIGTERM)
            try:
                code_rc = proc.wait(timeout=2)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
                code_rc = proc.wait()
    wall = time.perf_counter() - start
    run = {"mode": mode, "argv": cmd, "wall_seconds": wall, "returncode": code_rc, "timeout": timed_out}
    new = set(state.glob("*/*.json")) - old
    if new:
        if len(new) != 1:
            raise RuntimeError("ambiguous diagnostic records")
        rec = json.loads(new.pop().read_text())
        run["measurement"] = {k: rec.get(k) for k in ["measurement_id", "check_outcome", "collection_status", "collection_reason", "wall_ms", "exit_code", "signal", "output_degraded", "options", "metadata"]}
        snap = rec.get("events") or {}
        run["events"] = {k: snap.get(k) for k in ["status", "counts", "incomplete_count", "pending_build_count", "pending_package_count", "go_test_cache_hint_seen", "timings_truncated"]}
    result["runs"].append(run)
    (dest / "results.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    print(mode, code_rc, round(wall, 3), flush=True)
    if code_rc != 0:
        raise SystemExit("diagnostic failed; private stdout/stderr retained")
