func TestTask575RestartWarnings(t *testing.T) {
 y:=simpleBridgeYAML(".env","x.enc")+`stack:
  svc: {default_runner: script, runners: {script: {up: "true", stop: "true"}}}
plans:
  one: {entries: [{name: svc}]}
  both: {composes: [{plan: one}]}
`
 for _, a:=range [][]string{nil,{"one"},{"both"}} { f:=newBridgeFixture(t,y,map[string]string{"x.enc":"x"}); t.Chdir(f.cfg.FileDir()); withEnvPolicyGlobals(t,nil,false); var e error; out:=captureStderr(t,func(){e=restartCmd.RunE(restartCmd,a)}); t.Logf("args=%q err=%v warned=%t out=%q",a,e,strings.Contains(out,"declares sops_source"),out) }
 f:=newBridgeFixture(t,simpleBridgeYAML(".env","x.enc")+`stack: {svc: {default_runner: script, runners: {script: {up: "true", stop: "true"}}}}
`,map[string]string{"x.enc":"x"}); t.Chdir(f.cfg.FileDir()); withEnvPolicyGlobals(t,nil,false); var e error; out:=captureStderr(t,func(){e=restartCmd.RunE(restartCmd,nil)}); t.Logf("whole-stack err=%v warned=%t out=%q",e,strings.Contains(out,"declares sops_source"),out)
}
